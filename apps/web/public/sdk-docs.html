@media-sdk/react & @media-sdk/core

The media-core package is a pure TypeScript SDK for fetching media from the Pexels API, providing event tracking (view, download), request deduplication, and in-memory caching.

1. Provider Setup & Authentication

import { MediaProvider } from '@media-sdk/react';

function App() {
  return (
    <MediaProvider apiKey="YOUR_PEXELS_API_KEY">
      <YourApp />
    </MediaProvider>
  );
}

2. Fetching Media via React Hooks

import { useMediaSearch, useMediaVideoSearch } from '@media-sdk/react';

function PhotoFeed() {
  const { data, loading, error, hasMore, loadMore } = useMediaSearch('ocean', { perPage: 16 });

  if (loading && data.length === 0) return <p>Loading...</p>;

  return (
    <div>
      {data.map(item => (
        <img key={item.id} src={item.src.medium} alt={item.alt} />
      ))}
      {hasMore && <button onClick={loadMore}>Load More</button>}
    </div>
  );
}

3. Telemetry Event Emitter (`view`, `download`)

import { useMediaEvent, usePexelsClient } from '@media-sdk/react';

function EventLogger() {
  const client = usePexelsClient();

  useMediaEvent('view', (event) => {
    console.log('User viewed photo:', event.media.id);
  });

  useMediaEvent('download', (event) => {
    console.log('User downloaded photo:', event.media.id);
  });

  const handleDownload = (photo) => {
    client.emit('download', { media: photo, timestamp: Date.now() });
  };
}

4. Core Architectural Principles

  • Zero UI/DOM Leaks: media-core contains no React or DOM imports.
  • Request Deduplication: 5-minute in-memory cache layer to prevent redundant Pexels API network calls.
