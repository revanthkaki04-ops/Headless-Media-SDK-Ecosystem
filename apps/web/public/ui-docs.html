@media-sdk/ui-react

The media-ui-react package delivers pure headless UI prop-getter hooks with zero shipped styles. Consumers supply custom markup and CSS while the package manages infinite scrolling, focus trapping, vertical video snap paging, and keyboard shortcuts.

1. Headless Infinite Grid Hook (`useGrid`)

import { useGrid } from '@media-sdk/ui-react';

function GridExample({ data, fetchNextPage, hasNextPage, isLoading }) {
  const { getContainerProps, getItemProps, getSentinelProps } = useGrid({
    fetchNextPage,
    hasNextPage,
    isLoading
  });

  return (
    <div {...getContainerProps({ className: 'my-custom-grid' })}>
      {data.map((item, index) => (
        <div key={item.id} {...getItemProps(index, { className: 'grid-item' })}>
          <img src={item.src} alt={item.alt} />
        </div>
      ))}
      <div {...getSentinelProps()}>{isLoading && 'Loading...'}</div>
    </div>
  );
}

2. Focus-Trapped Accessible Lightbox (`useLightbox`)

import { useLightbox } from '@media-sdk/ui-react';

function LightboxExample({ totalItems, activeIndex }) {
  const { isOpen, open, close, next, prev, getOverlayProps, getContentProps, getCloseButtonProps } = useLightbox({
    totalItems
  });

  if (!isOpen) return null;

  return (
    <div {...getOverlayProps({ className: 'lightbox-modal' })}>
      <div {...getContentProps({ className: 'modal-body' })}>
        <button {...getCloseButtonProps()}>✕</button>
        <button onClick={prev}>←</button>
        <button onClick={next}>→</button>
      </div>
    </div>
  );
}

3. Throttled Vertical Reel Swiper (`useReelSwiper`)

import { useReelSwiper } from '@media-sdk/ui-react';

function VideoReelView({ videos }) {
  const { activeIndex, getContainerProps, getSlideProps } = useReelSwiper({
    totalItems: videos.length
  });

  return (
    <div {...getContainerProps({ className: 'reel-swiper' })}>
      <div className="reel-track" style={{ transform: `translateY(-${activeIndex * 100}%)` }}>
        {videos.map((video, index) => (
          <div key={video.id} {...getSlideProps(index, { className: 'reel-slide' })}>
            <video src={video.link} autoPlay={index === activeIndex} loop muted />
          </div>
        ))}
      </div>
    </div>
  );
}

4. Zero-Dependency Contract

  • Zero SDK Imports: media-ui-react has 0 imports from media-core or media-react.
  • Zero CSS Dependencies: Completely unstyled prop-getter architecture.
